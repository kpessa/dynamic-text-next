### Data Models
**Test Case Types** [New]
```typescript
interface TestCase {
  id: string;
  name: string;
  description?: string;
  category: 'boundary' | 'combinatorial' | 'edge' | 'manual' | 'template';
  inputs: TestInput[];
  expectedOutput: ExpectedOutput;
  assertions: Assertion[];
  priority: 'high' | 'medium' | 'low';
  tags: string[];
  enabled: boolean;
}

interface TestInput {
  variable: string;
  value: any;
  type: string;
  description?: string;
}

interface ExpectedOutput {
  type: 'exact' | 'contains' | 'regex' | 'range';
  value: any;
  tolerance?: number; // For numeric comparisons
}

interface Assertion {
  type: 'equals' | 'contains' | 'matches' | 'inRange' | 'truthy' | 'falsy';
  path?: string; // For nested object assertions
  expected: any;
  message?: string;
}

interface TestSuite {
  id: string;
  name: string;
  sectionId: string;
  testCases: TestCase[];
  coverage: CoverageMetrics;
  createdAt: Date;
  lastRun?: Date;
}

interface CoverageMetrics {
  statementCoverage: number;
  branchCoverage: number;
  conditionCoverage: number;
  pathCoverage: number;
  uncoveredLines: number[];
  uncoveredBranches: string[];
}
```

### Test Generation Strategies

**Boundary Value Analysis**:
```typescript
// For numeric variable with range [0, 100]
const boundaryTests = [
  { value: -1, expected: 'error' },     // Below min
  { value: 0, expected: 'valid' },      // Min boundary
  { value: 1, expected: 'valid' },      // Just above min
  { value: 50, expected: 'valid' },     // Typical value
  { value: 99, expected: 'valid' },     // Just below max
  { value: 100, expected: 'valid' },    // Max boundary
  { value: 101, expected: 'error' }     // Above max
];
```

**Pairwise Testing Example**:
```typescript
// Variables: age (child/adult), weight (low/normal/high)
const pairwiseTests = [
  { age: 'child', weight: 'low' },
  { age: 'child', weight: 'normal' },
  { age: 'adult', weight: 'high' },
  { age: 'adult', weight: 'low' }
  // Covers all pairs with minimal tests
];
```

### Coverage Calculation
```typescript
// Branch coverage example
const calculateBranchCoverage = (code, testCases) => {
  const branches = extractBranches(code);
  const coveredBranches = new Set();
  
  testCases.forEach(test => {
    const executedBranches = executeAndTrack(code, test);
    executedBranches.forEach(b => coveredBranches.add(b));
  });
  
  return (coveredBranches.size / branches.length) * 100;
};
```
